cmake_minimum_required(VERSION 3.21)
project(NebulaVision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets Sql Network Concurrent Multimedia MultimediaWidgets HttpServer)

# 与原pro共同保留；源码目录限定为正式模块，避免把临时测试入口编进客户端。
file(GLOB CLIENT_SOURCES CONFIGURE_DEPENDS model/*.cpp util/*.cpp service/*.cpp view/*.cpp player/*.cpp)
file(GLOB CLIENT_HEADERS CONFIGURE_DEPENDS model/*.h util/*.h service/*.h view/*.h player/*.h)
add_executable(class15 main.cpp resources.qrc ${CLIENT_SOURCES} ${CLIENT_HEADERS})
target_include_directories(class15 PRIVATE util)
target_link_libraries(class15 PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Sql Qt6::Network Qt6::Concurrent Qt6::Multimedia Qt6::MultimediaWidgets)
if(WIN32)
    set_target_properties(class15 PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
add_executable(nebulavision_server backend/main.cpp backend/apiserver.cpp backend/mediaadapter.cpp backend/recorder.cpp backend/redisclient.cpp backend/detectionjobs.cpp backend/apiserver.h)
target_link_libraries(nebulavision_server PRIVATE Qt6::Core Qt6::Sql Qt6::Network Qt6::Concurrent Qt6::HttpServer)
install(TARGETS class15 nebulavision_server RUNTIME DESTINATION bin)
install(DIRECTORY agent DESTINATION . PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY NebulaVision/docs DESTINATION NebulaVision)
